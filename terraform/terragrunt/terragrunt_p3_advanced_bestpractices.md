## 1. Generate Blocks Nâng Cao

### 1.1 Generate Provider Cho Multi-Subscription Azure

```hcl
# root terragrunt.hcl - hỗ trợ nhiều Azure subscriptions

locals {
  env_vars       = read_terragrunt_config(find_in_parent_folders("env.hcl"))
  subscription_id = local.env_vars.locals.subscription_id
  tenant_id       = get_env("ARM_TENANT_ID", "")
}

# Generate provider.tf tự động
generate "provider" {
  path      = "provider.tf"
  if_exists = "overwrite_terragrunt"
  contents  = <<EOF
terraform {
  required_version = ">= 1.7.0"
  required_providers {
    azurerm = {
      source  = "hashicorp/azurerm"
      version = "~> 3.90"
    }
    azuread = {
      source  = "hashicorp/azuread"
      version = "~> 2.47"
    }
    random = {
      source  = "hashicorp/random"
      version = "~> 3.6"
    }
  }
}

provider "azurerm" {
  features {
    resource_group {
      prevent_deletion_if_contains_resources = true
    }
  }
  subscription_id = "${local.subscription_id}"
  tenant_id       = "${local.tenant_id}"
}

provider "azuread" {
  tenant_id = "${local.tenant_id}"
}
EOF
}

# Generate versions.tf
generate "versions" {
  path      = "versions_lock.tf"
  if_exists = "overwrite_terragrunt"
  contents  = <<EOF
# Auto-generated by Terragrunt - DO NOT EDIT
# Managed in: root terragrunt.hcl
EOF
}
```

### 1.2 Generate Backend Config Dynamically

```hcl
# Hàm helper để tạo backend key tự động
locals {
  # Lấy relative path từ root
  # Ví dụ: live/production/networking → production/networking
  rel_path    = path_relative_to_include()
  path_parts  = split("/", local.rel_path)

  # Cấu trúc: live/<env>/<component>
  environment = local.path_parts[1]
  component   = local.path_parts[2]

  # Backend storage
  tf_state_rg  = "rg-terraform-state"
  tf_state_sa  = replace("tfstate${local.environment}", "-", "")  # lowercase, no hyphen
  tf_state_key = "${local.environment}/${local.component}.tfstate"
}

remote_state {
  backend = "azurerm"

  generate = {
    path      = "backend.tf"
    if_exists = "overwrite_terragrunt"
  }

  config = {
    resource_group_name  = local.tf_state_rg
    storage_account_name = local.tf_state_sa
    container_name       = "tfstate"
    key                  = local.tf_state_key
    use_azuread_auth     = true   # Dùng Azure AD thay vì storage key
  }
}
```

## 2. Before/After Hooks

```hcl
# terragrunt.hcl - Hooks cho mọi module

terraform {
  source = "../../../../modules//aks"

  # ===== BEFORE HOOKS =====

  # Kiểm tra Azure login
  before_hook "check_azure_login" {
    commands = ["apply", "plan", "destroy"]
    execute  = ["bash", "-c", "az account show &>/dev/null || (echo '❌ Not logged in to Azure!' && exit 1)"]
  }

  # Kiểm tra đúng subscription
  before_hook "check_subscription" {
    commands = ["apply", "destroy"]
    execute = [
      "bash", "-c",
      "CURRENT=$(az account show --query id -o tsv); EXPECTED=${local.subscription_id}; [ \"$CURRENT\" = \"$EXPECTED\" ] || (echo \"❌ Wrong subscription! Current: $CURRENT\" && exit 1)"
    ]
  }

  # Format check
  before_hook "fmt_check" {
    commands = ["apply", "plan"]
    execute  = ["terraform", "fmt", "-check", "-recursive"]
  }

  # Validate
  before_hook "validate" {
    commands = ["apply"]
    execute  = ["terraform", "validate"]
  }

  # ===== AFTER HOOKS =====

  # Notify sau khi apply thành công
  after_hook "notify_apply_success" {
    commands     = ["apply"]
    execute      = ["bash", "${get_repo_root()}/scripts/notify.sh", "apply", "${local.environment}", "${local.component}"]
    run_on_error = false
  }

  # Cleanup temp files
  after_hook "cleanup" {
    commands     = ["apply", "plan", "destroy"]
    execute      = ["bash", "-c", "rm -f tfplan *.tfplan 2>/dev/null || true"]
    run_on_error = true
  }

  # Backup state sau apply
  after_hook "backup_state" {
    commands     = ["apply"]
    execute = [
      "bash", "-c",
      "az storage blob snapshot --account-name ${local.tf_state_sa} --container-name tfstate --name ${local.tf_state_key}"
    ]
    run_on_error = true
  }
}
```

## 3. Dependency Patterns

### 3.1 Đọc Output từ Remote State

```hcl
# Cách 1: Terragrunt dependency (recommended)
dependency "networking" {
  config_path = "../networking"

  # Mock outputs cho plan khi networking chưa deploy
  mock_outputs_merge_strategy_with_state = "shallow"
  mock_outputs = {
    vnet_id    = "mock-vnet-id"
    subnet_ids = ["mock-subnet-1", "mock-subnet-2"]
  }
}

inputs = {
  vnet_id = dependency.networking.outputs.vnet_id
}

# Cách 2: data.terraform_remote_state (ít dùng với Terragrunt)
# Dùng khi cần output từ module không dùng Terragrunt
```

### 3.2 Cross-Environment Dependencies

```hcl
# production/app/terragrunt.hcl
# Muốn dùng shared resources từ _global env

dependency "global_dns" {
  config_path = "../../_global/dns"   # Cross-environment!
  mock_outputs = {
    zone_id   = "mock-zone-id"
    zone_name = "company.com"
  }
  mock_outputs_allowed_terraform_commands = ["validate", "plan"]
}

inputs = {
  dns_zone_id   = dependency.global_dns.outputs.zone_id
  dns_zone_name = dependency.global_dns.outputs.zone_name
}
```

## 4. Testing Terragrunt với Terratest

```go
// test/aks_test.go
package test

import (
    "testing"
    "github.com/gruntwork-io/terratest/modules/terraform"
    "github.com/stretchr/testify/assert"
    "os"
)

func TestAKSModule(t *testing.T) {
    t.Parallel()

    // Lấy credentials từ env
    subscriptionID := os.Getenv("ARM_SUBSCRIPTION_ID")

    terraformOptions := &terraform.Options{
        // Trỏ đến module cần test
        TerraformDir: "../modules/aks",

        // Variables
        Vars: map[string]interface{}{
            "project_name":        "test",
            "environment":         "test",
            "location":            "southeastasia",
            "resource_group_name": "rg-test-aks",
            "cluster_name":        "aks-test",
            "node_count":          1,
            "node_vm_size":        "Standard_D2s_v3",
            "kubernetes_version":  "1.28",
            "subnet_id":           "/subscriptions/.../subnets/snet-aks",
            "acr_id":              "/subscriptions/.../registries/testacr",
        },

        // Không dùng màu (clean output trong CI)
        NoColor: true,
    }

    // Cleanup sau test
    defer terraform.Destroy(t, terraformOptions)

    // Deploy
    terraform.InitAndApply(t, terraformOptions)

    // Verify outputs
    clusterName := terraform.Output(t, terraformOptions, "cluster_name")
    assert.Equal(t, "aks-test", clusterName)

    clusterID := terraform.Output(t, terraformOptions, "cluster_id")
    assert.Contains(t, clusterID, subscriptionID)

    // Verify AKS hoạt động
    kubeConfig := terraform.Output(t, terraformOptions, "kube_config")
    assert.NotEmpty(t, kubeConfig)
}
```

## 5. Terragrunt Best Practices

### 5.1 Folder Structure Pattern

```
infrastructure/
├── _envcommon/              ← Shared configs giữa envs
│   ├── aks.hcl              ← Common AKS config
│   ├── networking.hcl
│   └── acr.hcl
│
├── terragrunt.hcl           ← Root: backend, provider
│
└── live/
    ├── _global/             ← Cross-env resources
    │   ├── dns/
    │   └── certificates/
    │
    ├── staging/
    │   ├── env.hcl
    │   ├── networking/
    │   │   └── terragrunt.hcl  ← include _envcommon/networking.hcl
    │   └── aks/
    │       └── terragrunt.hcl  ← include _envcommon/aks.hcl
    │
    └── production/
        ├── env.hcl
        ├── networking/
        │   └── terragrunt.hcl  ← Same structure, diff values
        └── aks/
            └── terragrunt.hcl
```

```hcl
# _envcommon/aks.hcl - Shared AKS config
locals {
  env_vars = read_terragrunt_config(find_in_parent_folders("env.hcl"))
  env      = local.env_vars.locals
}

dependency "networking" {
  config_path = "../networking"
  mock_outputs = {
    resource_group_name     = "mock-rg"
    resource_group_location = "southeastasia"
    aks_subnet_id           = "mock-subnet"
  }
}

terraform {
  source = "${get_repo_root()}//modules/aks"
}

inputs = {
  resource_group_name = dependency.networking.outputs.resource_group_name
  location            = dependency.networking.outputs.resource_group_location
  subnet_id           = dependency.networking.outputs.aks_subnet_id
  kubernetes_version  = local.env.aks_k8s_version
  node_count          = local.env.aks_node_count
  node_vm_size        = local.env.aks_node_vm_size
}
```

```hcl
# live/staging/aks/terragrunt.hcl - Chỉ kế thừa + thêm override
include "root" {
  path = find_in_parent_folders()
}

include "aks_common" {
  path   = find_in_parent_folders("_envcommon/aks.hcl")
  expose = true

dependency "acr" {
  config_path = "../acr"
  mock_outputs = {
    acr_id = "mock-acr-id"
  }

# Override/thêm inputs riêng cho staging
inputs = {
  acr_id       = dependency.acr.outputs.acr_id
  cluster_name = "aks-myapp-staging"
}
```

### 5.2 Checklist Terragrunt

```
SETUP:
✅ Một root terragrunt.hcl cho backend + provider
✅ env.hcl cho mỗi environment (staging/production)
✅ _envcommon/ cho shared module configs
✅ Backend state lưu trên Azure Storage (riêng từng env)
✅ Blob versioning bật cho state recovery

SECURITY:
✅ Separate Azure subscriptions cho staging/production
✅ Service Principal riêng cho mỗi environment
✅ Không lưu credentials trong code (dùng env vars)
✅ State storage sử dụng Azure AD auth (use_azuread_auth=true)

CI/CD:
✅ Plan chạy tự động trên mọi PR
✅ Apply chỉ chạy sau khi merge vào main
✅ Manual approval gate cho production
✅ Artifacts plan lưu lại (plan → apply đảm bảo consistency)
✅ Separate pipeline jobs cho từng environment

MODULES:
✅ Dùng double-slash (//): source = "../../modules//aks"
✅ Mock outputs cho tất cả dependencies
✅ Semantic versioning cho shared modules
✅ Tests với Terratest cho modules quan trọng
```
